Project siga_its_modelo_unificado {
  database_type: "PostgreSQL"
  Note: '''
    SIGA ITS - Modelo unificado Gonzalo + Profesor.

    Criterios principales:
    - Se prioriza la normalización y la trayectoria académica por alumno + plan.
    - La autenticación/roles se delega a Directus (u otro proveedor externo).
      personas.usuario_uuid permite vincular la persona académica con el usuario externo.
    - "cursadas" representa la comisión/oferta de una materia; la inscripción del alumno
      se registra aparte en inscripciones_cursadas.
    - Exámenes se modelan como TURNO -> LLAMADO -> MESA.
    - Actas de cursada y actas de mesa permanecen separadas por tener reglas distintas,
      pero comparten documentos_actas, firmas e historial de auditoría.
    - El historial académico NO se duplica en una tabla: debe construirse mediante una VIEW
      a partir de actas, equivalencias e inscripciones.
    - Algunas reglas de negocio complejas (correlativas circulares, cambio de plan,
      consistencia del tipo de período, cierre de actas, etc.) deben reforzarse
      mediante backend, triggers o constraints SQL específicas.
  '''
}

  created_at timestamptz [not null, default: `now()`]
}


Table planes_carreras {
  id int [pk, increment]
  carrera_id int [not null]

  codigo varchar(30) [not null]
  nombre varchar(150) [not null]
  resolucion varchar(100) [null]

  duracion_anios int [not null]
  fecha_inicio_vigencia date [not null]
  fecha_fin_vigencia date [null]

  vigente_para_ingresantes boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]

  checks {
    `duracion_anios > 0` [name: 'duracion_plan_valida']
    `fecha_fin_vigencia IS NULL OR fecha_fin_vigencia > fecha_inicio_vigencia` [name: 'vigencia_plan_valida']
  }

  indexes {
    (carrera_id, codigo) [unique]
  }

  Note: '''
    Debe existir como máximo un plan "vigente_para_ingresantes = true"
    por carrera. En PostgreSQL conviene reforzarlo con un índice UNIQUE parcial.
  '''
}

Ref: planes_carreras.carrera_id > carreras.id

  checks {
    `plan_materia_id <> plan_materia_requerida_id` [name: 'correlativa_no_puede_ser_ella_misma']

  indexes {
    (plan_materia_id, plan_materia_requerida_id, requerida_para) [unique]
  }

  Note: '''
    Reglas a validar fuera de este diagrama:
    1) Ambas materias deben pertenecer al mismo plan.
    2) No deben existir ciclos A -> B -> A.
  '''

Ref: correlativas.plan_materia_requerida_id > planes_materias.id

Table cursadas {
  id int [pk, increment]

  sede_id int [not null]
  plan_materia_id int [not null]
  ciclo_lectivo int [not null]
  periodo periodo_cursada [not null]

  codigo varchar(30) [null]
  descripcion varchar(150) [null]

  porcentaje_asistencia_minima numeric(5,2) [not null, default: 75.00]
  nota_minima_regular numeric(4,2) [not null, default: 4.00]
  permite_promocion boolean [not null, default: true]
  nota_minima_promocion numeric(4,2) [null, default: 7.00]
  meses_vigencia_regularidad int [null]

  activo boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]

  checks {
    `ciclo_lectivo >= 2000` [name: 'ciclo_lectivo_cursada_valido']
    `porcentaje_asistencia_minima >= 0 AND porcentaje_asistencia_minima <= 100` [name: 'asistencia_minima_valida']
    `nota_minima_regular >= 0 AND nota_minima_regular <= 10` [name: 'nota_regular_valida']
    `nota_minima_promocion IS NULL OR (nota_minima_promocion >= 0 AND nota_minima_promocion <= 10)` [name: 'nota_promocion_valida']
  }

  indexes {
    (plan_materia_id, ciclo_lectivo, periodo, codigo) [unique]
  }

  Note: '''
    Equivale conceptualmente a una comisión/oferta de cursada.
    NO contiene al alumno. Los alumnos se vinculan mediante inscripciones_cursadas.
  '''
}

Ref: cursadas.plan_materia_id > planes_materias.id
Ref: cursadas.sede_id > sedes.id

  aula varchar(80) [null, note: ''' 
    Agregando las sedes podria ser conveniente agregar otra tabla de aulas pero por ahora no 
    lo considero necesario.

  activo boolean [not null, default: true]

  checks {
    `dia_semana BETWEEN 1 AND 7` [name: 'dia_semana_cursada_valido']
    `hora_fin > hora_inicio` [name: 'horario_cursada_valido']
  }

  indexes {
    (cursada_id, dia_semana, hora_inicio) [unique]
  }
}

  Esta tabla podria llenar a partir de un trigger que se ejecute a partir de un udpate en la tabla

/*
  1) CORRELATIVAS
     - validar mismo plan;
     - impedir ciclos;
     - validar requisitos al inscribirse a cursada o mesa.

  2) PLANES
     - máximo un plan vigente para ingresantes por carrera;
     - al cambiar de plan, cerrar estado de trayectoria anterior y crear nueva;
     - registrar cambios_plan.

  3) PERÍODOS
     - periodos_inscripcion_planes solo con período tipo carrera;
     - periodos_inscripcion_cursadas solo con período tipo cursada;
     - periodos_inscripcion_mesas solo con período tipo mesa.

  4) ACTAS
     - una vez cerrada un acta no debe modificarse directamente;
     - una corrección debe generar rectificación/auditoría;
     - documento_acta.tipo debe coincidir con la tabla académica relacionada;
     - los detalles solo pueden incluir inscripciones de la cursada/mesa correspondiente.

  5) MATERIA ACREDITADA
     Antes de confirmar una inscripción a mesa o una equivalencia:
     - verificar que la materia no esté ya aprobada por examen;
     - verificar que no esté promocionada;
     - verificar que no exista equivalencia aprobada equivalente.

  6) PREINSCRIPCIÓN
     - al confirmar, validar documentación/requisitos obligatorios;
     - crear alumno si todavía no existe;
     - crear alumnos_carreras y su estado inicial.

  7) HISTORIAL ACADÉMICO
     - implementar como VIEW;
     - si el rendimiento lo exige, usar MATERIALIZED VIEW con refresco controlado.
